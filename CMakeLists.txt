cmake_minimum_required(VERSION 3.16)
if(POLICY CMP0091)
  cmake_policy(SET CMP0091 NEW)
endif()
project(WinPuttier VERSION 0.4.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "WinPuttier is a native Windows application")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(winputtier_core STATIC native/core.cpp native/core.h)
target_compile_definitions(winputtier_core PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET winputtier_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(winputtier_core PUBLIC winhttp shell32 ole32 advapi32)

add_executable(WinPuttier WIN32 native/main.cpp native/preview.cpp native/preview.h)
target_link_libraries(WinPuttier PRIVATE winputtier_core user32 gdi32 comdlg32 comctl32)
target_compile_definitions(WinPuttier PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET WinPuttier PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(WinPuttier PROPERTIES OUTPUT_NAME WinPuttier)

include(CTest)
if(BUILD_TESTING)
  add_executable(winputtier_tests native/test_core.cpp)
  target_link_libraries(winputtier_tests PRIVATE winputtier_core)
  target_compile_definitions(winputtier_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_property(TARGET winputtier_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME core_reg_parsing COMMAND winputtier_tests)
endif()
